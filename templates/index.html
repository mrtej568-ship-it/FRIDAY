// BLUETOOTH PAKKA CONNECT - v206 let connectedDevice = null; let deviceName = localStorage.getItem('dh2_bt_device')||null; async function connectBluetooth(){ let chat = document.getElementById('chat'); addMsg('friday','Boss Bluetooth scanning... Phone lo Bluetooth ON chey + Visible pettu!',true); // Step 1: Check support if(!navigator.bluetooth){ addMsg('friday','Chrome browser lo open chey Boss! Firefox/WhatsApp browser lo Bluetooth work avadu. Chrome lo https:// nee site open chesi malli nokku. Kani call/sms links ippatike ready!',true); // Fallback - tel: still works return; } try{ // Step 2: PAKKA CONNECT - Filters tho addMsg('friday','Scanning nearby phones... Phone peru select chey list lo',true); let device = await navigator.bluetooth.requestDevice({ acceptAllDevices: true, optionalServices: ['battery_service','device_information','00001800-0000-1000-8000-00805f9b34fb','00001801-0000-1000-8000-00805f9b34fb'] }); connectedDevice = device; deviceName = device.name; localStorage.setItem('dh2_bt_device', device.name); addMsg('friday',`✅ CONNECTED TO ${device.name.toUpperCase()}! Boss!`,true); document.getElementById('voice-name').innerText = `BT: ${device.name.toUpperCase()} ✓`; // Step 3: Connect to GATT try{ let server = await device.gatt.connect(); addMsg('friday',`🔋 GATT Connected! Battery info chaduvuthunna...`,true); // Try battery try{ let service = await server.getPrimaryService('battery_service'); let char = await service.getCharacteristic('battery_level'); let value = await char.readValue(); addMsg('friday',`📱 ${device.name} Battery: ${value.getUint8(0)}% - Link pakka active Boss!`,true); }catch(e){ addMsg('friday',`📱 ${device.name} linked! Ippudu call/sms chesthe nee phone ki direct ga velthundi!`,true); } }catch(e){ addMsg('friday',`✅ Device paired: ${device.name} - GATT optional, kaani pairing success! Call/SMS ready!`,true); } // Step 4: Auto disconnect handle device.addEventListener('gattserverdisconnected', ()=>{ addMsg('friday',`⚠️ ${device.name} disconnected Boss. Malli CONNECT button nokku.`,true); document.getElementById('voice-name').innerText = `BT: DISCONNECTED`; connectedDevice = null; }); }catch(err){ if(err.name==='NotFoundError'){ addMsg('friday','❌ No device selected. Malli CONNECT nokki phone select chey Boss.',true); }else if(err.name==='NotAllowedError'){ addMsg('friday','❌ Permission block. Phone lo Bluetooth ON + Chrome lo Location ON pettu Boss.',true); }else{ addMsg('friday',`Bluetooth Error: ${err.message}
Kaani tel: sms: links pakka work avthunnayi - call button nokku, phone dialer open avthundi!`,true); } } } // AUTO SHOW CONNECTED STATUS ON LOAD setTimeout(()=>{ if(deviceName){ addMsg('friday',`Last connected: ${deviceName} - Malli connect avvali ante CONNECT PHONE nokku Boss.`,true); } },1000); // CALL/SMS - PAKKA PHONE KI VELTHUNDI function makeCall(name){ let c=JSON.parse(localStorage.getItem('dh2_contacts')||'[]').find(x=>x.name.toLowerCase()===name.toLowerCase()); if(!c||!c.phone){ addMsg('friday',`${name} number ledu. Contact add chey.`,true); return; } addMsg('friday',`📞 Calling ${c.name} via ${connectedDevice?connectedDevice.name:'Phone'} - Dialer open avthundi...`,true); // Pakka phone ki vellali ante - tel: protocol window.location.href=`tel:${c.phone}`; } function makeSMS(name,text){ let c=JSON.parse(localStorage.getItem('dh2_contacts')||'[]').find(x=>x.name.toLowerCase()===name.toLowerCase()); if(!c||!c.phone){ addMsg('friday',`${name} number ledu.`,true); return; } let m=text||prompt(`Message to ${c.name}:`); if(!m) return; addMsg('friday',`💬 SMS to ${c.name} via phone...`,true); window.location.href=`sms:${c.phone}?body=${encodeURIComponent(m)}`; }
